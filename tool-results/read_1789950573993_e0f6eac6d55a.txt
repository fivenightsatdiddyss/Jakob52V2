   130→// Renders an avatar circle based on the avatar encoding.
   131→// size: 'sm' (28px sidebar/roster) | 'md' (36px messages) | 'lg' (48px editor preview)
   132→function Avatar({
   133→  avatar,
   134→  name,
   135→  size = 'md',
   136→  className,
   137→}: {
   138→  avatar: string
   139→  name: string
   140→  size?: 'sm' | 'md' | 'lg'
   141→  className?: string
   142→}) {
   143→  const dims =
   144→    size === 'sm'
   145→      ? 'h-7 w-7 text-xs'
   146→      : size === 'lg'
   147→        ? 'h-12 w-12 text-base'
   148→        : 'h-9 w-9 text-sm'
   149→  const initial = initialOf(name)
   150→
   151→  if (avatar.startsWith('preset:')) {
   152→    return (
   153→      <div
   154→        className={cn(
   155→          'grid shrink-0 place-items-center rounded-full bg-gradient-to-br font-bold text-white shadow-inner',
   156→          dims,
   157→          presetGradient(avatar),
   158→          className,
   159→        )}
   160→      >
   161→        {initial}
   162→      </div>
   163→    )
   164→  }
   165→
   166→  if (isUrl(avatar)) {
   167→    return (
   168→      <img
   169→        src={avatar}
   170→        alt={name ? `${name} avatar` : 'avatar'}
   171→        className={cn('shrink-0 rounded-full object-cover', dims, className)}
   172→      />
   173→    )
   174→  }
   175→
   176→  if (isEmoji(avatar)) {
   177→    return (
   178→      <div
   179→        className={cn(
   180→          'grid shrink-0 place-items-center rounded-full bg-white/10 ring-1 ring-inset ring-white/15',
   181→          dims,
   182→          className,
   183→        )}
   184→      >
   185→        <span className="leading-none">
   186→          {size === 'lg' ? 'text-2xl' : size === 'sm' ? 'text-base' : 'text-lg'}
   187→          {avatar}
   188→        </span>
   189→      </div>
   190→    )
   191→  }
   192→
   193→  // fallback → initial in a plain circle
   194→  return (
   195→    <div
   196→      className={cn(
   197→        'grid shrink-0 place-items-center rounded-full bg-white/10 font-bold text-white/80 ring-1 ring-inset ring-white/15',
   198→        dims,
   199→        className,
   200→      )}
   201→    >
   202→      {initial}
   203→    </div>
   204→  )
   205→}

   207→// Downscale an image file to a small data URL (≤ AVATAR_MAX chars).
   208→// Tries progressively smaller sizes / lower qualities until it fits.
   209→const downscaleToDataUrl = (file: File): Promise<string> =>
   210→  new Promise((resolve, reject) => {
   211→    if (!file.type.startsWith('image/')) {
   212→      reject(new Error('not an image'))
   213→      return
   214→    }
   215→    const reader = new FileReader()
   216→    reader.onerror = () => reject(new Error('read failed'))
   217→    reader.onload = () => {
   218→      const img = new Image()
   219→      img.onerror = () => reject(new Error('decode failed'))
   220→      img.onload = () => {
   221→        const minDim = Math.min(img.width, img.height) || 1
   222→        const sx = (img.width - minDim) / 2
   223→        const sy = (img.height - minDim) / 2
   224→        const sizes = [64, 48, 40, 32, 28, 24]
   225→        const qualities = [0.7, 0.5, 0.35, 0.25]
   226→        for (const size of sizes) {
   227→          for (const q of qualities) {
   228→            const canvas = document.createElement('canvas')
   229→            canvas.width = size
   230→            canvas.height = size
   231→            const ctx = canvas.getContext('2d')
   232→            if (!ctx) continue
   233→            ctx.drawImage(img, sx, sy, minDim, minDim, 0, 0, size, size)
   234→            const url = canvas.toDataURL('image/jpeg', q)
   235→            if (url.length <= AVATAR_MAX) {
   236→              resolve(url)
   237→              return
   238→            }
   239→          }
   240→        }
   241→        reject(new Error('too large'))
   242→      }
   243→      img.src = reader.result as string
   244→    }
   245→    reader.readAsDataURL(file)
   246→  })

   272→  const [uploading, setUploading] = useState(false)
   273→
   274→  const scrollRef = useRef<HTMLDivElement>(null)
   275→  const socketRef = useRef<Socket | null>(null)
   276→  const typingClearRef = useRef<ReturnType<typeof setTimeout> | null>(null)

   278→  // ---- load profile from localStorage on mount ----

   299→                : 'preset:0',
   300→          }
   301→        }
   302→      }
   303→    } catch {
   304→      loaded = null
   305→    }
   306→    if (!loaded) {
   307→      loaded = { name: randomGuest(), color: pickColor(), avatar: 'preset:0' }
   308→    }
   309→    setProfile(loaded)
   310→    setProfileLoaded(true)
   311→  }, [])
   312→
   313→  // ---- socket lifecycle ----
   314→  useEffect(() => {
   315→    // CRITICAL: gateway requires path "/" + ?XTransformPort=3003 in the URI.
   316→    // Baking the query into the URI string (matching the websocket demo) ensures
   317→    // socket.io uses "/" as the engine.io path so Caddy can route to port 3003.
   318→    // Never use a direct http://localhost:3003 URL — that bypasses the gateway.
   319→    const socket = io('/?XTransformPort=3003', {
   320→      transports: ['websocket', 'polling'],
   321→      reconnection: true,
   322→      reconnectionAttempts: Infinity,
   323→      reconnectionDelay: 1000,
   324→      reconnectionDelayMax: 5000,
   325→    })
   326→    socketRef.current = socket
   327→
   328→    socket.on('connect', () => {
   329→      setConnected(true)
   330→      // announce our profile so the server roster knows who we are
   331→      const p = profileRef.current
   332→      if (p.name) {
   333→        socket.emit('profile', { user: p.name, color: p.color, avatar: p.avatar })
   334→      }
   335→    })
   336→    socket.on('disconnect', () => setConnected(false))
   337→    socket.on('presence', (p: PresencePayload) => setOnline(p.online))
   338→    socket.on('history', (hist: ChatMsg[]) =>
   339→      setMessages(Array.isArray(hist) ? hist : []),
   340→    )
   341→    socket.on('message', (msg: ChatMsg) =>
   342→      setMessages((prev) =>
