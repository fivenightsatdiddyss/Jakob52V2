   145→// Renders an avatar circle based on the avatar encoding.
   146→// size: 'sm' (28px sidebar/roster) | 'md' (36px messages) | 'lg' (48px editor preview)
   147→function Avatar({
   148→  avatar,
   149→  name,
   150→  size = 'md',
   151→  className,
   152→}: {
   153→  avatar: string
   154→  name: string
   155→  size?: 'sm' | 'md' | 'lg'
   156→  className?: string
   157→}) {
   158→  const dims =
   159→    size === 'sm'
   160→      ? 'h-7 w-7 text-xs'
   161→      : size === 'lg'
   162→        ? 'h-12 w-12 text-base'
   163→        : 'h-9 w-9 text-sm'
   164→  const initial = initialOf(name)
   165→
   166→  if (avatar.startsWith('preset:')) {
   167→    return (
   168→      <div
   169→        className={cn(
   170→          'grid shrink-0 place-items-center rounded-full bg-gradient-to-br font-bold text-white shadow-inner',
   171→          dims,
   172→          presetGradient(avatar),
   173→          className,
   174→        )}
   175→      >
   176→        {initial}
   177→      </div>
   178→    )
   179→  }
   180→
   181→  if (isUrl(avatar)) {
   182→    return (
   183→      <img
   184→        src={avatar}
   185→        alt={name ? `${name} avatar` : 'avatar'}
   186→        className={cn('shrink-0 rounded-full object-cover', dims, className)}
   187→      />
   188→    )
   189→  }
   190→
   191→  if (isEmoji(avatar)) {
   192→    return (
   193→      <div
   194→        className={cn(
   195→          'grid shrink-0 place-items-center rounded-full bg-white/10 ring-1 ring-inset ring-white/15',
   196→          dims,
   197→          className,
   198→        )}
   199→      >
   200→        <span className="leading-none">
   201→          {size === 'lg' ? 'text-2xl' : size === 'sm' ? 'text-base' : 'text-lg'}
   202→          {avatar}
   203→        </span>
   204→      </div>
   205→    )
   206→  }
   207→
   208→  // fallback → initial in a plain circle
   209→  return (
   210→    <div
   211→      className={cn(
   212→        'grid shrink-0 place-items-center rounded-full bg-white/10 font-bold text-white/80 ring-1 ring-inset ring-white/15',
   213→        dims,
   214→        className,
   215→      )}
   216→    >
   217→      {initial}
   218→    </div>
   219→  )
   220→}

   222→// Downscale an image file to a small data URL (≤ AVATAR_MAX chars).
   223→// Tries progressively smaller sizes / lower qualities until it fits.
   224→const downscaleToDataUrl = (file: File): Promise<string> =>
   225→  new Promise((resolve, reject) => {
   226→    if (!file.type.startsWith('image/')) {
   227→      reject(new Error('not an image'))
   228→      return
   229→    }
   230→    const reader = new FileReader()
   231→    reader.onerror = () => reject(new Error('read failed'))
   232→    reader.onload = () => {
   233→      const img = new Image()
   234→      img.onerror = () => reject(new Error('decode failed'))
   235→      img.onload = () => {
   236→        const minDim = Math.min(img.width, img.height) || 1
   237→        const sx = (img.width - minDim) / 2
   238→        const sy = (img.height - minDim) / 2
   239→        const sizes = [64, 48, 40, 32, 28, 24]
   240→        const qualities = [0.7, 0.5, 0.35, 0.25]
   241→        for (const size of sizes) {
   242→          for (const q of qualities) {
   243→            const canvas = document.createElement('canvas')
   244→            canvas.width = size
   245→            canvas.height = size
   246→            const ctx = canvas.getContext('2d')
   247→            if (!ctx) continue
   248→            ctx.drawImage(img, sx, sy, minDim, minDim, 0, 0, size, size)
   249→            const url = canvas.toDataURL('image/jpeg', q)
   250→            if (url.length <= AVATAR_MAX) {
   251→              resolve(url)
   252→              return
   253→            }
   254→          }
   255→        }
   256→        reject(new Error('too large'))
   257→      }
   258→      img.src = reader.result as string
   259→    }
   260→    reader.readAsDataURL(file)
   261→  })

   278→  const [typingUser, setTypingUser] = useState<string | null>(null)

   281→  // profile editor modal state
   282→  const [editing, setEditing] = useState(false)

   284→  const [draftColor, setDraftColor] = useState('')

   287→  const [uploading, setUploading] = useState(false)

   289→  const scrollRef = useRef<HTMLDivElement>(null)

   291→  const lastMessageTimeRef = useRef<string>('')
   292→  const lastPollSuccessRef = useRef<number>(0)
   293→  const lastPresenceRef = useRef<number>(0)
   294→  const typingThrottleRef = useRef<number>(0)
   295→  const typingClearRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   296→  const pollTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   297→  const mountedRef = useRef<boolean>(true)
   298→
   299→  // ---- POST helper (best-effort, fire-and-forget) ----
   300→  const postChat = (body: Record<string, unknown>) => {
   301→    if (!mountedRef.current) return
   302→    void fetch('/api/chat', {
   303→      method: 'POST',
   304→      headers: { 'content-type': 'application/json' },
   305→      body: JSON.stringify(body),
   306→      keepalive: true,
   307→    }).catch(() => {
   308→      /* swallow — polling will reconcile */
   309→    })
   310→  }
   311→

   313→    const sid = sessionIdRef.current
   314→    if (!sid || !p.name) return
   315→    postChat({
   316→      op: 'presence',
   317→      sessionId: sid,
   318→      user: p.name,
   319→      color: p.color,
   320→      avatar: p.avatar,
   321→    })
   322→    lastPresenceRef.current = Date.now()
   323→  }
   324→
   325→  // ---- load profile + session on mount, start polling ----
   326→  useEffect(() => {
   327→    if (typeof window === 'undefined') return
   328→    mountedRef.current = true
   329→
   330→    // load profile from localStorage (or generate a fresh guest)
   331→    let loaded: Profile | null = null
   332→    try {
   333→      const raw = window.localStorage.getItem(LS_KEY)
   334→      if (raw) {
   335→        const parsed = JSON.parse(raw) as Partial<Profile>
   336→        if (parsed && typeof parsed === 'object') {
   337→          loaded = {
   338→            name:
   339→              typeof parsed.name === 'string'
   340→                ? parsed.name.slice(0, 24).trim() || randomGuest()
   341→                : randomGuest(),
   342→            color:
   343→              typeof parsed.color === 'string' && parsed.color
   344→                ? parsed.color.slice(0, 40)
   345→                : pickColor(),
   346→            avatar:
   347→              typeof parsed.avatar === 'string' && parsed.avatar
   348→                ? parsed.avatar.slice(0, AVATAR_MAX)

   353→    } catch {
   354→      loaded = null

   356→    if (!loaded) {
   357→      loaded = { name: randomGuest(), color: pickColor(), avatar: 'preset:0' }
   358→    }
   359→    setProfile(loaded)
   360→    profileRef.current = loaded
   361→    setProfileLoaded(true)
   362→
   363→    // stable session id — persists across refresh within the same browser tab
   364→    let sid = ''
   365→    try {
   366→      sid = window.sessionStorage.getItem(SESSION_KEY) || ''
   367→      if (!sid) {
   368→        sid = `s-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 10)}`
   369→        window.sessionStorage.setItem(SESSION_KEY, sid)
   370→      }
   371→    } catch {
   372→      sid = `s-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 10)}`
   373→    }
   374→    sessionIdRef.current = sid
   375→
   376→    // initial presence so the roster sees us immediately
   377→    sendPresence(loaded)
   378→
   379→    // ---- polling loop ----
   380→    const poll = async () => {
   381→      if (!mountedRef.current) return
   382→      const sid2 = sessionIdRef.current
   383→      const p = profileRef.current
   384→      const since = lastMessageTimeRef.current
   385→      try {
   386→        const url = `/api/chat?since=${encodeURIComponent(since)}`
   387→        const res = await fetch(url, { cache: 'no-store' })
   388→        if (!res.ok) throw new Error(`http ${res.status}`)
   389→        const data = (await res.json()) as PollResponse
   390→        if (!mountedRef.current) return
   391→        lastPollSuccessRef.current = Date.now()
   392→        setConnected(true)
   393→
   394→        // merge messages — dedupe by id; replace optimistic local messages
   395→        const newMsgs = Array.isArray(data.messages) ? data.messages : []
   396→        if (newMsgs.length > 0) {
   397→          setMessages((prev) => {
   398→            const result = [...prev]
   399→            for (const server of newMsgs) {
   400→              if (result.some((m) => m.id === server.id)) continue
   401→              const serverT = new Date(server.time).getTime()
   402→              const localIdx = result.findIndex(
   403→                (m) =>
   404→                  m.id.startsWith(OPTIMISTIC_PREFIX) &&
   405→                  m.user === server.user &&
   406→                  m.text === server.text &&
   407→                  Math.abs(new Date(m.time).getTime() - serverT) < 5000,
   408→              )
   409→              if (localIdx >= 0) result[localIdx] = server
   410→              else result.push(server)
   411→            }
   412→            const latest = result[result.length - 1]
   413→            if (latest) {
   414→              const latestT = new Date(latest.time).getTime()
   415→              const curT = new Date(lastMessageTimeRef.current).getTime()
   416→              if (Number.isNaN(curT) || latestT > curT) {
   417→                lastMessageTimeRef.current = latest.time
   418→              }
   419→            }
   420→            return result.slice(-100)
   421→          })
   422→        }
   423→
   424→        if (typeof data.online === 'number') setOnline(data.online)
   425→        if (Array.isArray(data.roster)) setRoster(data.roster)
   426→
   427→        // typing indicator — only show someone else typing
   428→        const typingArr = Array.isArray(data.typing) ? data.typing : []
   429→        const other = typingArr.find((t) => t.sessionId !== sid2)
   430→        setTypingUser(other ? other.user : null)
   431→        if (other && typingClearRef.current == null) {
   432→          typingClearRef.current = setTimeout(() => {
   433→            setTypingUser(null)
   434→            typingClearRef.current = null
   435→          }, TYPING_CLIENT_TIMEOUT_MS)
   436→        } else if (!other && typingClearRef.current) {
   437→          clearTimeout(typingClearRef.current)
   438→          typingClearRef.current = null
   439→        }
   440→
   441→        // presence keepalive — piggyback on the polling loop
   442→        if (Date.now() - lastPresenceRef.current > PRESENCE_KEEPALIVE_MS) {
   443→          sendPresence(p)
   444→        }
   445→      } catch {
   446→        if (!mountedRef.current) return
   447→        const stale = Date.now() - lastPollSuccessRef.current > CONNECT_STALE_MS
   448→        setConnected(!stale)
   449→      }
   450→    }
   451→    void poll()
   452→    pollTimerRef.current = setInterval(() => {
   453→      void poll()
   454→    }, POLL_INTERVAL_MS)
   455→
   456→    return () => {
   457→      mountedRef.current = false
   458→      if (pollTimerRef.current) clearInterval(pollTimerRef.current)
   459→      pollTimerRef.current = null
   460→      if (typingClearRef.current) clearTimeout(typingClearRef.current)
   461→      typingClearRef.current = null
   462→      // best-effort: clear typing indicator so others don't see us typing forever
   463→      const sid2 = sessionIdRef.current
   464→      const p = profileRef.current
   465→      if (sid2 && p.name) {
   466→        void fetch('/api/chat', {
   467→          method: 'POST',
   468→          headers: { 'content-type': 'application/json' },
   469→          body: JSON.stringify({
   470→            op: 'typing',
   471→            sessionId: sid2,
   472→            user: p.name,
   473→            typing: false,
   474→          }),
   475→          keepalive: true,
   476→        }).catch(() => {
   477→          /* swallow */
   478→        })
   479→      }
   480→    }
   481→  }, [])
   482→
   483→  // re-broadcast presence whenever the profile changes (so the roster
   484→  // updates live with the new name/color/avatar)
   485→  useEffect(() => {
   486→    if (!profileLoaded) return
   487→    sendPresence(profile)
   488→  }, [profile, profileLoaded])
   489→
   490→  // ---- auto-scroll on new message ----
   491→  useEffect(() => {
   492→    const el = scrollRef.current
   493→    if (el) el.scrollTo({ top: el.scrollHeight, behavior: 'smooth' })
   494→  }, [messages.length])
   495→
   496→  // ---- profile editor helpers ----
   497→  const openEditor = () => {
   498→    setDraftName(profile.name)
   499→    setDraftColor(profile.color)
   500→    setDraftAvatar(profile.avatar)
   501→    if (profile.avatar.startsWith('preset:')) setAvatarMode('preset')
   502→    else if (isUrl(profile.avatar)) setAvatarMode('upload')
   503→    else setAvatarMode('emoji')
   504→    setEditing(true)
   505→  }
   506→
   507→  const saveProfile = () => {
   508→    const name = draftName.trim().slice(0, 24) || randomGuest()
   509→    const color = draftColor || pickColor()
   510→    const avatar = draftAvatar || 'preset:0'
   511→    const next: Profile = { name, color, avatar }
   512→    setProfile(next)
   513→    profileRef.current = next
   514→    try {
   515→      window.localStorage.setItem(LS_KEY, JSON.stringify(next))
   516→    } catch {
   517→      /* ignore quota errors */
   518→    }
   519→    setEditing(false)
   520→    toast.success('Profile saved', {
   521→      description: `You are now ${name}`,
   522→    })
   523→  }
   524→
   525→  const handleUpload = async (file: File) => {
   526→    setUploading(true)
   527→    try {
   528→      const url = await downscaleToDataUrl(file)
   529→      setDraftAvatar(url)
   530→      setAvatarMode('upload')
   531→      toast.success('Avatar ready')
   532→    } catch (err) {
   533→      const msg = err instanceof Error ? err.message : 'failed'
   534→      if (msg === 'too large') {
   535→        toast.error("Couldn't shrink image enough", {
   536→          description: 'Try a smaller or simpler image',
   537→        })
   538→      } else if (msg === 'not an image') {
   539→        toast.error('Please choose an image file')
   540→      } else {
   541→        toast.error('Could not load image')
   542→      }
   543→    } finally {
   544→      setUploading(false)
   545→    }
   546→  }
   547→
   548→  // ---- messaging ----
   549→  const send = () => {
   550→    const value = input.trim().slice(0, 500)
   551→    if (!value) return
   552→    const p = profileRef.current
   553→    if (!p.name) return
   554→    // optimistic append — the next poll will reconcile/replace with the server copy
   555→    const optimistic: ChatMsg = {
   556→      id: `${OPTIMISTIC_PREFIX}${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 6)}`,
   557→      user: p.name,
   558→      text: value,
   559→      color: p.color,
   560→      avatar: p.avatar,
   561→      time: new Date().toISOString(),
   562→    }
   563→    setMessages((prev) => [...prev, optimistic])
   564→    postChat({
   565→      op: 'message',
   566→      user: p.name,
   567→      text: value,
   568→      color: p.color,
   569→      avatar: p.avatar,
   570→    })
   571→    // clear typing indicator
   572→    if (typingClearRef.current) clearTimeout(typingClearRef.current)
   573→    typingClearRef.current = null
   574→    setTypingUser(null)
   575→    const sid = sessionIdRef.current
   576→    if (sid) {
   577→      postChat({ op: 'typing', sessionId: sid, user: p.name, typing: false })
   578→    }
   579→    setInput('')
   580→  }
   581→
   582→  const onInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   583→    setInput(e.target.value)
   584→    const p = profileRef.current
   585→    const sid = sessionIdRef.current
   586→    if (!sid || !p.name) return
